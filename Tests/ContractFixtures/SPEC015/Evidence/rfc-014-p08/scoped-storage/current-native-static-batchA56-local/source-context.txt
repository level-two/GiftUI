    var path: InlineArray<20, UInt8> = .init(repeating: 0)
    var plan: InlineArray<112, UInt8> = .init(repeating: 0)
    var measurement: CounterGeometryMeasurement {
        .init(layout: counterLayoutLimits, render: counterRenderLimits, renderWorkspace: counterRenderCapacity, drawing: counterDrawingLimits,
            layoutBytes: UInt64(scopes.count + text.count + MemoryLayout<StaticCounterGeometry>.size - MemoryLayout<QualifiedRender>.size - MemoryLayout<StaticRecordDrawingWorkspace<StaticCounterDrawingWorkspace>>.size),
            renderBytes: UInt64(MemoryLayout<QualifiedRender>.size), pathBytes: UInt64(path.count),
            planBytes: UInt64(plan.count + MemoryLayout<StaticRecordDrawingWorkspace<StaticCounterDrawingWorkspace>>.size))
    }
    mutating func withGeometry(_ body: (consuming StaticCounterGeometry) -> Int) -> Int {
        withUnsafeMutableBytes(of: &scopes) { s in
            withUnsafeMutableBytes(of: &text) { t in
                withUnsafeMutableBytes(of: &path) { p in
                    withUnsafeMutableBytes(of: &plan) { d in
                        body(StaticCounterGeometry(scopes: s, text: t, path: p, plan: d))
                    }
                }
            }
        }
    }

func counterValidateBacking<G, C, R, H, GR: CounterStorageRegionSource>(profile: RuntimeProfileKind, limits: RuntimeProfileLimits,
    root: CounterRootLayout, rootMetadataRegion: CounterStorageRegion, semantic: CounterSemanticMeasurement, geometry: CounterGeometryMeasurement,
    interaction: borrowing InteractionOwnedKeyState<C, R, H>, geometryType: G.Type,
    canvasCount: Int, canvasCapacity: UInt16, canvasControlBytes: Int,
    factoryRegion: CounterStorageRegion,
    factoryDescriptorRegion: CounterStorageRegion? = nil,
    geometryRegions: borrowing GR,
    report: (CounterStorageRegion) -> Void) -> RuntimeProfileValidationResult
where G: CounterGeometry, C: CounterCandidateBacking & InteractionOwnedBankMeasurementSource, R: CounterCommittedBacking & InteractionOwnedBankMeasurementSource, H: CounterHitBacking & InteractionOwnedBankMeasurementSource,
    C.Identity == SemanticCompleteIdentity, R.Identity == SemanticCompleteIdentity, H.Identity == SemanticCompleteIdentity {
    let facts = interaction.ownedStorageMeasurement()
    let metadata = MemoryLayout<InteractionOwnedKeyState<C, R, H>>.size - MemoryLayout<C>.size - 2 * MemoryLayout<R>.size - 2 * MemoryLayout<H>.size
    func physical<Record>(_ bank: InteractionOwnedBankMeasurement, record: Record.Type) -> CounterPhysicalBankFacts {
        .init(bank: bank, recordSize: bank.retainedPayloadBytes == 0 ? MemoryLayout<Record?>.size : MemoryLayout<Record>.size, recordStride: bank.retainedPayloadBytes == 0 ? MemoryLayout<Record?>.stride : MemoryLayout<Record>.stride, recordAlignment: bank.retainedPayloadBytes == 0 ? MemoryLayout<Record?>.alignment : MemoryLayout<Record>.alignment,
            optionalRecords: bank.retainedPayloadBytes == 0, retainedBytes: bank.inlineSize + bank.retainedPayloadBytes)
    }
    let rows = CounterPhysicalInteractionFacts(
        candidate: physical(facts.candidate, record: InteractionCandidateRecord<SemanticCompleteIdentity>.self),
        candidateHits: physical(facts.candidateHits, record: InteractionHitRegion<SemanticCompleteIdentity>.self),
        prepared: physical(facts.prepared, record: BoundActionRecord<SemanticCompleteIdentity>.self),
        committed: physical(facts.committed, record: BoundActionRecord<SemanticCompleteIdentity>.self),
        committedHits: physical(facts.committedHits, record: InteractionHitRegion<SemanticCompleteIdentity>.self), stateMetadataBytes: metadata)
    return counterValidatePhysicalBacking(profile: profile, limits: limits, root: root, rootMetadataRegion: rootMetadataRegion,
        semantic: semantic, geometry: geometry, interactionFacts: rows, identityCapacities: interaction.identityCapacities,
        geometryType: geometryType, candidateType: C.self, committedType: R.self, hitType: H.self,
        canvasCount: canvasCount, canvasCapacity: canvasCapacity, canvasControlBytes: canvasControlBytes,
        factoryRegion: factoryRegion, factoryDescriptorRegion: factoryDescriptorRegion, geometryRegions: geometryRegions, report: report)
}

func counterValidatePhysicalBacking<G, C, R, H, GR: CounterStorageRegionSource>(profile: RuntimeProfileKind, limits: RuntimeProfileLimits,
    root: CounterRootLayout, rootMetadataRegion: CounterStorageRegion, semantic: CounterSemanticMeasurement, geometry: CounterGeometryMeasurement,
    interactionFacts: CounterPhysicalInteractionFacts, identityCapacities: (perKey: UInt16, perBank: UInt32),
    descriptorBytes: Int = 0, geometryType: G.Type, candidateType: C.Type, committedType: R.Type, hitType: H.Type,
    canvasCount: Int, canvasCapacity: UInt16, canvasControlBytes: Int,
    factoryRegion: CounterStorageRegion,
    factoryDescriptorRegion: CounterStorageRegion? = nil,
    geometryRegions: borrowing GR,
    report: (CounterStorageRegion) -> Void) -> RuntimeProfileValidationResult
where G: CounterGeometry & ~Copyable & ~Escapable, C: CounterCandidateBacking & ~Copyable & ~Escapable, R: CounterCommittedBacking & ~Copyable & ~Escapable, H: CounterHitBacking & ~Copyable & ~Escapable,
    C.Identity == SemanticCompleteIdentity, R.Identity == SemanticCompleteIdentity, H.Identity == SemanticCompleteIdentity {
    // The concrete scoped render adapter exports exactly the configured view;
    // inaccessible physical headroom remains measured in the backing regions.
    guard counterRenderWorkspaceCovers(geometry.renderWorkspace, limits.renderWorkspace) else { return .invalid(.insufficientStorage) }
    let keys = interactionFacts.measurement
    let keyLimits = identityCapacities
    let semanticLimits = semantic.limits
    guard let recordBytes = semantic.records.bytes, let pathBytes = semantic.paths.bytes,
        let payloadBytes = semantic.payloads.bytes, let scalarBytes = semantic.scalars.bytes,
        let canvasBytes = CounterExtent.elements(CanvasStagedEntry<SemanticRecordReference, CounterValueTable.CaptureStorage>?.self, count: canvasCount).bytes else { return .invalid(.arithmeticOverflow) }
    let interactionMetadata = interactionFacts.stateMetadataBytes
    let pending = MemoryLayout<InlineArray<2, NormalizedPointerEvent?>>.size + MemoryLayout<InlineArray<34, UInt8>>.size + MemoryLayout<UInt16>.size + MemoryLayout<UInt16?>.size
    let sealed = MemoryLayout<CounterAdmissionBacking>.size - pending
    let result = MemoryLayout<RuntimeCompletePipelineResult<RuntimeOwnerFailure>?>.size
    typealias Admission = CounterAdmission
    typealias Owner = CounterPipelineOwner<G, C, R, H>
    typealias Coordinator = StaticRuntimeExecutionCoordinator<CounterExecutionBinding, Admission, Owner>
    var concreteCoordinatorBytes = MemoryLayout<Coordinator>.size
    #if !hasFeature(Embedded)
    if profile == .dynamic {
        concreteCoordinatorBytes = MemoryLayout<DynamicRuntimeExecutionCoordinator<CounterExecutionBinding, Admission, Owner>>.size
    }
    #endif
    // Endpoint/sink data now resides in Control but still belongs to the backend
    // accounting owner. Subtract it once from the combined coordinator/control.
    // Physical backend arrays in CounterControl remain fully charged below.
    let coordinator = concreteCoordinatorBytes - MemoryLayout<G>.size
        - MemoryLayout<InteractionOwnedKeyState<C, R, H>>.size - MemoryLayout<CounterEndpoint>.size
        + MemoryLayout<CounterControl>.size - Admission.pointerStateStorageBytes
        - MemoryLayout<CounterAdmissionBacking>.size - MemoryLayout<CounterPointerState>.size - result
        + descriptorBytes + Int(factoryRegion.chargedBytes) + Int(factoryDescriptorRegion?.chargedBytes ?? 0)
    // Bank zero occupies exactly the configured whole-element prefix. Bank one
    // owns the second configured slice plus any retained allocator tail; never
    // divide a typed element when Array capacity rounds to an odd count.
    let firstRecords = CounterExtent.elements(SemanticRuntimeRecord.self, count: Int(semanticLimits.records)).bytes!
    let firstPaths = CounterExtent.elements(SemanticPathEntry.self, count: Int(semanticLimits.paths)).bytes!
    let firstPayloads = CounterExtent.elements(CounterPayload?.self, count: Int(semanticLimits.payloads)).bytes!
    let firstScalars = CounterExtent.elements(UInt32.self, count: Int(semanticLimits.textScalars)).bytes!
    guard semantic.records.count >= Int(semanticLimits.records) * 2,
        semantic.paths.count >= Int(semanticLimits.paths) * 2,
        semantic.payloads.count >= Int(semanticLimits.payloads) * 2,
        semantic.scalars.count >= Int(semanticLimits.textScalars) * 2 else { return .invalid(.insufficientStorage) }
    let candidatePayload = firstPayloads + firstScalars
    let publishedPayload = payloadBytes - firstPayloads + scalarBytes - firstScalars
    let firstMatch = UInt64(semanticLimits.paths) * UInt64(MemoryLayout<UInt16>.size)
    let allMatch = UInt64(semantic.paths.count) * UInt64(MemoryLayout<UInt16>.size)
    let matchControl = 4 * MemoryLayout<UInt32>.size
    guard firstPaths >= firstMatch, pathBytes >= allMatch,
        semantic.controlBytes >= matchControl, coordinator >= 0,
        interactionMetadata >= 0 else { return .invalid(.invariantViolation) }
    let values: InlineArray<26, UInt64> = [firstPaths - firstMatch, pathBytes - firstPaths - (allMatch - firstMatch), candidatePayload, publishedPayload,
        allMatch + UInt64(matchControl), keys.candidateIdentityBytes, keys.committedIdentityBytes,
        UInt64(root.metadataBytes), UInt64(root.modelBytes), UInt64(root.pinBytes),
        firstRecords + UInt64(semantic.controlBytes - matchControl), recordBytes - firstRecords,
        geometry.layoutBytes, geometry.renderBytes, canvasBytes + UInt64(canvasControlBytes), geometry.pathBytes, geometry.planBytes,
        UInt64(root.liveBytes), UInt64(root.candidateBytes), keys.candidateBytes + UInt64(interactionMetadata), keys.committedBytes,
        UInt64(pending), UInt64(sealed), UInt64(Admission.pointerStateStorageBytes + MemoryLayout<CounterPointerState>.size), UInt64(coordinator), UInt64(result)]
    for index in 0..<values.count where UInt32(exactly: values[index]) == nil { return .invalid(.arithmeticOverflow) }
    var reported: InlineArray<26, UInt64> = .init(repeating: 0)
    var reportOverflow = false
    func record(_ region: CounterStorageRegion) {
        precondition((0..<26).contains(region.family))
        let sum = reported[region.family].addingReportingOverflow(region.chargedBytes)
        reportOverflow = reportOverflow || sum.overflow
        reported[region.family] = sum.partialValue
        report(region)
    }
    record(.elements(SemanticPathEntry.self, count: Int(semanticLimits.paths), family: 0, chargedBytes: values[0]))
    record(.elements(SemanticPathEntry.self, count: semantic.paths.count - Int(semanticLimits.paths), family: 1, chargedBytes: values[1]))
    record(.elements(CounterPayload?.self, count: Int(semanticLimits.payloads), family: 2))
    record(.elements(UInt32.self, count: Int(semanticLimits.textScalars), family: 2))
    record(.elements(CounterPayload?.self, count: semantic.payloads.count - Int(semanticLimits.payloads), family: 3))
    record(.elements(UInt32.self, count: semantic.scalars.count - Int(semanticLimits.textScalars), family: 3))
    // Repeated embedded fields, not a contiguous allocation or another bank.
    record(.elements(UInt16.self, count: semantic.paths.count, family: 4))
    record(.elements(UInt32.self, count: 4, family: 4))
    record(rootMetadataRegion)
    record(.elements(CounterOwnedModel?.self, count: Int(root.models), family: 8))
    record(.elements(Bool.self, count: Int(root.pins), family: 9))
    record(.elements(SemanticRuntimeRecord.self, count: Int(semanticLimits.records), family: 10))
    if profile == .static {
        record(.inline(CounterSemanticBacking.self, family: 10, chargedBytes: CounterSemanticBacking.paddingBytes, kind: .metadataPadding))
        record(.inline(StaticSemanticRecordStorage.self, family: 10, chargedBytes: MemoryLayout<StaticSemanticRecordStorage>.size - matchControl, kind: .partition))
    }
    #if !hasFeature(Embedded)
    if profile == .dynamic {
        record(.inline(UInt16.self, family: 7))
        record(.elements(Bool.self, count: 2, family: 7))
        record(.inline(DynamicObservableModelStorage<CounterOwnedModel, CounterModelSchema>.self, family: 7))
        record(.inline(InlineArray<2, SemanticRecordBankMetadata>.self, family: 10, chargedBytes: MemoryLayout<InlineArray<2, SemanticRecordBankMetadata>>.size - matchControl, kind: .partition))
        record(.inline([SemanticRuntimeRecord].self, family: 10))
        record(.inline([SemanticPathEntry].self, family: 10))
        record(.inline([CounterPayload?].self, family: 10))
        record(.inline([UInt32].self, family: 10))
        record(.inline(SemanticRecordLimits.self, family: 10))
        record(.inline(SemanticRecordOwnerIdentity.self, family: 10))
        record(.inline(Bool.self, family: 10))
    }
    #endif
    record(.elements(SemanticRuntimeRecord.self, count: semantic.records.count - Int(semanticLimits.records), family: 11))
    geometryRegions.visitStorageRegions(record)
    record(.elements(CanvasStagedEntry<SemanticRecordReference, CounterValueTable.CaptureStorage>?.self, count: canvasCount, family: 14))
    if profile == .static { record(.inline(StaticCanvasValueStorage.self, family: 14)) }
    #if !hasFeature(Embedded)
    if profile == .dynamic { record(.inline(DynamicCanvasCallableStorage<SemanticRecordReference, CounterValueTable>.self, family: 14)) }
    #endif
    record(.inline(ObservableStateRootStorage<SemanticCompleteIdentity>.self, family: 17, chargedBytes: root.liveBytes, kind: .partition))
    record(.inline(ObservableStateRootStorage<SemanticCompleteIdentity>.self, family: 18, chargedBytes: root.candidateBytes, kind: .partition))
    interactionFacts.visitStorageRegions(record)
    record(.elements(NormalizedPointerEvent?.self, count: 2, family: 21))
    record(.inline(InlineArray<34, UInt8>.self, family: 21))
    record(.inline(UInt16.self, family: 21))
    record(.inline(UInt16?.self, family: 21))
    record(.inline(CounterAdmissionBacking.self, family: 22, chargedBytes: sealed, kind: .partition))
    record(.inline(Admission.Controller?.self, family: 23, chargedBytes: Admission.pointerStateStorageBytes, kind: .partition))
    record(.inline(CounterPointerState.self, family: 23))
    let coordinatorFields = concreteCoordinatorBytes - MemoryLayout<G>.size - MemoryLayout<InteractionOwnedKeyState<C, R, H>>.size
    if profile == .static { record(.inline(Coordinator.self, family: 24, chargedBytes: coordinatorFields + descriptorBytes, kind: .metadataPadding)) }
    #if !hasFeature(Embedded)
    if profile == .dynamic { record(.inline(DynamicRuntimeExecutionCoordinator<CounterExecutionBinding, Admission, Owner>.self, family: 24, chargedBytes: coordinatorFields + descriptorBytes, kind: .metadataPadding)) }
    #endif
    // All outer controls, observations, optional controller metadata and padding
    // remain physically live and are charged here exactly once.
    record(.inline(CounterControl.self, family: 24, chargedBytes: MemoryLayout<CounterControl>.size - MemoryLayout<CounterEndpoint>.size - Admission.pointerStateStorageBytes - MemoryLayout<CounterAdmissionBacking>.size - MemoryLayout<CounterPointerState>.size - result, kind: .metadataPadding))
    record(factoryRegion)
    if let factoryDescriptorRegion { record(factoryDescriptorRegion) }
    record(.inline(RuntimeCompletePipelineResult<RuntimeOwnerFailure>?.self, family: 25))
    guard !reportOverflow else { return .invalid(.arithmeticOverflow) }
    for index in 0..<26 { precondition(reported[index] == values[index], "physical region audit mismatch") }
    let bytes = RuntimeStorageByteCounts(semanticCandidatePathBytes: UInt32(values[0]), semanticPublishedPathBytes: UInt32(values[1]), semanticCandidatePayloadBytes: UInt32(values[2]), semanticPublishedPayloadBytes: UInt32(values[3]), identityMatchWorkspaceBytes: UInt32(values[4]), interactionCandidateIdentityBytes: UInt32(values[5]), interactionCommittedIdentityBytes: UInt32(values[6]), modelMetadataBytes: UInt32(values[7]), modelOwnerBytes: UInt32(values[8]), framePinBytes: UInt32(values[9]), semanticCandidateBytes: UInt32(values[10]), semanticPublishedBytes: UInt32(values[11]), layoutCandidateBytes: UInt32(values[12]), renderWorkspaceBytes: UInt32(values[13]), canvasCallableBytes: UInt32(values[14]), pathWorkspaceBytes: UInt32(values[15]), drawingPlanBytes: UInt32(values[16]), observableLiveBytes: UInt32(values[17]), observableCandidateBytes: UInt32(values[18]), interactionCandidateBytes: UInt32(values[19]), interactionCommittedBytes: UInt32(values[20]), admissionQueueBytes: UInt32(values[21]), sealedBatchBytes: UInt32(values[22]), pointerStateBytes: UInt32(values[23]), coordinatorStateBytes: UInt32(values[24]), failureStateBytes: UInt32(values[25]))
    let records = RuntimeRecordLimits(maximumPathEntries: semanticLimits.paths, maximumPathDepth: semanticLimits.depth, maximumPathChildFanout: semanticLimits.fanout, maximumPayloadBytesPerBank: semanticLimits.payloadBytes, maximumIdentityComponentsPerKey: keyLimits.perKey, maximumInteractionIdentityComponents: keyLimits.perBank)
    let inputs = RuntimeProfileLimitInputs(records: limits.records, models: limits.models, semantic: limits.semantic, maximumSemanticStructuralOccurrences: limits.maximumSemanticStructuralOccurrences, layout: limits.layout, render: limits.render, renderWorkspace: limits.renderWorkspace, renderSink: limits.renderSink, maximumOrdinaryRenderOperations: limits.maximumOrdinaryRenderOperations, execution: limits.execution, observableState: limits.observableState, interaction: limits.interaction, drawing: limits.drawing, staticCanvas: limits.staticCanvas, profile: profile)
    let rootCapacity = ObservableStateRootStorage<SemanticCompleteIdentity>.storageCapacities
    let capacities = RuntimeStorageCapacities(candidateRecords: records, publishedRecords: records,
        models: .init(maximumModelSlots: root.models, maximumOwningTickets: root.claims, maximumFramePins: root.pins, maximumModelCaptureBanks: root.captures),
        semanticCandidate: .init(maximumDepth: semanticLimits.depth, maximumSemanticNodes: semanticLimits.records, maximumBodyEvaluations: semanticLimits.records, maximumModifierApplications: semanticLimits.records, maximumActionOccurrences: semanticLimits.records),
        semanticPublished: .init(maximumDepth: semanticLimits.depth, maximumSemanticNodes: semanticLimits.records, maximumBodyEvaluations: semanticLimits.records, maximumModifierApplications: semanticLimits.records, maximumActionOccurrences: semanticLimits.records),
        semanticCandidateStructuralOccurrences: semanticLimits.records, semanticPublishedStructuralOccurrences: semanticLimits.records,
        layoutCandidate: geometry.layout, render: geometry.render, renderWorkspace: limits.renderWorkspace,
        canvasCallableOccurrences: canvasCapacity, staticCanvasCaptureBytes: UInt16(MemoryLayout<CounterValueTable.CaptureStorage>.stride),
        pathPoints: geometry.drawing.maximumLivePathPoints, pathSubpaths: geometry.drawing.maximumLivePathSubpaths,
        drawingPlanStrokes: geometry.drawing.maximumPlanStrokes, drawingPlanPoints: geometry.drawing.maximumPlanPoints, drawingPlanSubpaths: geometry.drawing.maximumPlanSubpaths, normalizedStrokeOperations: geometry.drawing.maximumNormalizedStrokeOperations,
        observableLiveLocations: rootCapacity.locations, observableLiveRegistrations: rootCapacity.registrations, observableCandidateAssociations: rootCapacity.associations,
        interactionCandidateActions: keys.candidateRecords, interactionCandidateHitRegions: keys.candidateHits, interactionCommittedActions: keys.committedRecords, interactionCommittedHitRegions: keys.committedHits,
        admissionQueue: counterExecutionLimits, sealedBatch: counterExecutionLimits, pointerStates: 1,
        coordinatorStatePresent: concreteCoordinatorBytes > 0, failureStatePresent: result > 0, byteCounts: bytes)
    if profile == .static { return RuntimeProfileValidator.validateStatic(inputs: inputs, capacities: capacities, metadata: CounterCanvasAudit()) }
    return RuntimeProfileValidator.validateDynamic(inputs: inputs, capacities: capacities)
}

#if !hasFeature(Embedded)
typealias CounterDynamicCandidate = DynamicInteractionCandidateStorage<SemanticCompleteIdentity>
typealias CounterDynamicCommitted = DynamicInteractionCommittedStorage<SemanticCompleteIdentity>

struct DynamicCounterProfileStorage: ~Copyable, RuntimeProfileStorage {
    typealias StructuralIdentity = SemanticCompleteIdentity
    static var profile: RuntimeProfileKind { .dynamic }
    let limits: RuntimeProfileLimits
    var root: DynamicObservableRootAdapter<CounterOwnedModel, SemanticCompleteIdentity>
    var semantic: DynamicSemanticRecordStorage<CounterPayload>
    var canvas = DynamicCanvasCallableStorage<SemanticRecordReference, CounterValueTable>(capacity: 1)
    var geometry: DynamicCounterGeometry? = DynamicCounterGeometry()
    var interaction: CounterDynamicKeys?
    var control = CounterControl()
    var used = false
    init(modelOwner: ObservableModelOwnerIdentity, semanticOwner: SemanticRecordOwnerIdentity, limits: RuntimeProfileLimits = counterActualLimits(.dynamic)) {
        self.limits = limits
        root = DynamicObservableRootAdapter(identity: modelOwner, schema: 1, needsCapture: false, modelLimit: min(2, max(1, limits.models.maximumModelSlots)), claimLimit: min(4, max(1, limits.models.maximumOwningTickets)), pinLimit: min(2, max(1, limits.models.maximumFramePins)), captureLimit: 0)!
        semantic = DynamicSemanticRecordStorage(limits: counterSemanticLimits, owner: semanticOwner)
        interaction = CounterDynamicKeys(state: InteractionState(candidateRecords: CounterDynamicCandidate(capacity: 2), candidateHitRegions: CounterDynamicHits(capacity: 2), candidateCommittedRecords: CounterDynamicCommitted(capacity: 2), committedRecords: CounterDynamicCommitted(capacity: 2), committedHitRegions: CounterDynamicHits(capacity: 2)), maximumComponentsPerKey: 24, maximumComponents: 48)!
    }
    borrowing func audit() -> RuntimeProfileValidationResult { audit(report: { _ in }) }
    borrowing func audit(report: (CounterStorageRegion) -> Void) -> RuntimeProfileValidationResult {
        guard let geometry, let interaction else { return .invalid(.missingStorage) }
        let extent = semantic.retainedExtents
        return counterValidateBacking(profile: Self.profile, limits: limits, root: root.storageLayout, rootMetadataRegion: .inline(ObservableModelOwnership.self, family: 7, chargedBytes: root.storageLayout.metadataBytes - MemoryLayout<UInt16>.size - 2 * MemoryLayout<Bool>.size - MemoryLayout<DynamicObservableModelStorage<CounterOwnedModel, CounterModelSchema>>.size, kind: .partition),
            semantic: .init(limits: semantic.storageLimits, records: .elements(SemanticRuntimeRecord.self, count: extent.records), paths: .elements(SemanticPathEntry.self, count: extent.paths), payloads: .elements(CounterPayload?.self, count: extent.payloads), scalars: .elements(UInt32.self, count: extent.scalars), controlBytes: extent.controlBytes),
            geometry: geometry.storageMeasurement(), interaction: interaction, geometryType: DynamicCounterGeometry.self,
            canvasCount: canvas.retainedElementCapacity, canvasCapacity: canvas.capacity,
            canvasControlBytes: MemoryLayout<DynamicCanvasCallableStorage<SemanticRecordReference, CounterValueTable>>.size,
            factoryRegion: .inline(Self.self, family: 24, chargedBytes: MemoryLayout<Self>.size - MemoryLayout<DynamicCanvasCallableStorage<SemanticRecordReference, CounterValueTable>>.size - MemoryLayout<DynamicCounterGeometry>.size - MemoryLayout<CounterDynamicKeys>.size - MemoryLayout<CounterControl>.size, kind: .metadataPadding),
            geometryRegions: geometry, report: report)
    }
    mutating func resetAttemptStorage() {
        precondition(geometry != nil && interaction != nil)
        geometry!.reset()
        interaction!.resolveCandidate(.discard)
        control.admission.finishAttempt()
    }
    mutating func resetAllStorage() {
        control.requireVacantRuntimeSlot()
        precondition(!used || control.unavailable)
        resetAttemptStorage()
        interaction!.resetAllStorage()
        control.admission.resetAll()
    }

    mutating func run() -> Int {
        guard !used, case .valid(let audited) = audit() else { return 0 }
        used = true
        precondition(interaction!.configureIdentityLimits(perKey: limits.records.maximumIdentityComponentsPerKey, perBank: limits.records.maximumInteractionIdentityComponents))
        var movedGeometry = geometry.take()
        var movedInteraction = interaction.take()
        var returned: CounterReturnedWorkspaces<DynamicCounterGeometry, CounterDynamicCandidate, CounterDynamicCommitted, CounterDynamicHits>?
        do throws(ObservableStateError) {
            try root.withLifetime(limits: limits.observableState) { (models, root) throws(ObservableStateError) in
                do throws(SemanticExpansionError) {
                    try semantic.withLifetime(configuredLimits: counterSemanticConfiguration(limits)) { (semantic) throws(SemanticExpansionError) in
                        do throws(DrawingError) {
                            try canvas.withLifetime(table: CounterValueTable(), limits: limits.staticCanvas ?? counterCanvasLimits, configuredCapacity: limits.drawing.maximumCanvasOccurrences) { values in
                                returned = exerciseCounterCoordinator(models: models, root: root, semantic: semantic, values: values, dynamic: true, geometry: movedGeometry.take()!, interaction: movedInteraction.take()!, audit: audited, control: &control)
                            }
                        } catch { preconditionFailure("factory Canvas") }
                    }
                } catch { preconditionFailure("factory semantic") }
            }
        } catch { preconditionFailure("factory model") }
        let workspaces = returned.take()!
        let checks = workspaces.checks
        geometry = workspaces.geometry; interaction = workspaces.interaction
        precondition(audit() == .valid(audited))
        resetAllStorage()
        precondition(audit() == .valid(audited))
        return checks + 6
    }
}

        try models.validateOperation(discovery: discovery)

            guard models.phase == .discovering else { throw .invalidPhaseSafetyNotProven }

            records.pointee.candidateActive = true
            for i in 0 ..< 2 { records.pointee.records[i].encountered = false }
            return .success(.candidateStarted)
        } catch { return .failure(error) }

    @inlinable

        for i in 0 ..< 2
        where identities[i] == identity
            && records.pointee.records[i].ordinal == ordinal
        { return i }
        return nil
    }

        let record = records.pointee.records[slot]
        return _GiftUIModelKey(
            ownerInstance: record.keyOwner, allocationGeneration: record.keyGeneration,
            slot: UInt16(slot))

        guard let slot = matching(identity, ordinal: ordinal),
            records.pointee.records[slot].published
        else { return nil }

    package borrowing func generation(
        identity: Identity, ordinal: UInt16, publishable: Bool = false

        guard let slot = matching(identity, ordinal: ordinal) else { return nil }
        let record = records.pointee.records[slot]
        if publishable {
            guard records.pointee.candidateActive, record.encountered else { return nil }
        } else if !record.published {
            return nil

        return record.attachment?.generation

    @inlinable
    internal borrowing func bind(_ state: inout State<Model.Schema>, slot: Int)
        throws(ObservableStateError)
    {
        let record = records.pointee.records[slot]
        guard let attachment = record.attachment else { throw .invariantViolation }
        let binding = _GiftUIBoundState(
            locationSlot: UInt16(slot), declarationOrdinal: record.ordinal,
            registrationGeneration: attachment.generation, key: key(at: slot))
        if let prior = state._giftUIBinding {
            guard prior.locationSlot == binding.locationSlot,
                prior.declarationOrdinal == binding.declarationOrdinal,
                prior.registrationGeneration == binding.registrationGeneration,
                prior.key == binding.key
            else {
                throw .staleModel
            }
        } else {
            state._giftUIBind(binding)
        }
    }

    @inlinable
    borrowing func preserved(
        _ identity: Identity, ordinal: UInt16, state: inout State<Model.Schema>
    ) throws(ObservableStateError) -> Bool {
        try checkOperation(discovery: true)
        guard models.phase == .discovering, records.pointee.candidateActive else {
            throw .invalidPhaseSafetyNotProven
        }
        guard let slot = matching(identity, ordinal: ordinal) else { return false }
        try models.pin(key(at: slot))
        try bind(&state, slot: slot)
        records.pointee.records[slot].encountered = true
        return true
    }

    @inlinable
    @inline(__always)
    package borrowing func encounter(
        identity: Identity, ordinal: UInt16, state: inout State<Model.Schema>
    ) -> ObservableStateResult {
        do throws(ObservableStateError) {
            if try preserved(identity, ordinal: ordinal, state: &state) {
                return .success(.preserved)
            }
            if case .candidate(let key)? = state._giftUIInitial {
                for i in 0 ..< 4 {
                    guard let claim = records.pointee.offeredClaims[i],
                        models.key(for: claim.address) == key
                    else { continue }
                    var ticket = ObservableModelTicket<Model.Schema>(claim: claim, lifetime: models)
                    try install(identity, ordinal: ordinal, state: &state, ticket: &ticket)
                    records.pointee.offeredClaims[i] = nil
                    return .success(.materialized)
                }
                throw .incompatibleAssociation
            }
            guard case .seed(let seed)? = state._giftUIInitial else {
                throw .incompatibleAssociation
            }
            // A fresh typed seed has no prior owner to duplicate. Generation
            // exhaustion precedes every model metadata/backing capacity check.
            guard records.pointee.nextGeneration != nil else {
                throw .registrationGenerationExhausted
            }
            var ticket = try models.materialize(Model.materialize(seed))
            do {
                try install(identity, ordinal: ordinal, state: &state, ticket: &ticket)
                return .success(.materialized)
            } catch {
                try? models.release(&ticket)
                throw error
            }
        } catch { return .failure(error) }
    }

    /// Explicitly consumes the host's matching candidate claim after attachment
    /// succeeds. On failure it remains the caller's ticket and model.
    @inlinable
    @inline(__always)
    package borrowing func encounter(
        identity: Identity, ordinal: UInt16, state: inout State<Model.Schema>,
        transferring ticket: inout ObservableModelTicket<Model.Schema>
    ) -> ObservableStateResult {
        do throws(ObservableStateError) {
            if try preserved(identity, ordinal: ordinal, state: &state) {
                return .success(.preserved)
            }
            guard case .candidate(let descriptor)? = state._giftUIInitial, descriptor == ticket.key
            else {
                throw .incompatibleAssociation
            }
            try models.validateTicket(ticket)
            try install(identity, ordinal: ordinal, state: &state, ticket: &ticket)
            return .success(.materialized)
        } catch { return .failure(error) }
    }

    /// Transfers an explicit claim into this activation's bounded candidate set.
    /// Unused claims are released on candidate completion or activation teardown.
    package borrowing func offerCandidate(_ ticket: inout ObservableModelTicket<Model.Schema>)
        -> ObservableStateResult
    {
        do throws(ObservableStateError) {
            try checkOperation()
            try models.validateTicket(ticket)
            for i in 0 ..< 4 where records.pointee.offeredClaims[i] == nil {
                records.pointee.offeredClaims[i] = try ticket.requireClaim()
                ticket.endOwnership()
                return .success(.unchanged)
            }
            throw .ownerCapacityExhausted
        } catch { return .failure(error) }
    }

    private borrowing func releaseOffers() throws(ObservableStateError) {
        var failure: ObservableStateError?
        for i in 0 ..< 4 {
            guard let claim = records.pointee.offeredClaims[i] else { continue }
            var ticket = ObservableModelTicket<Model.Schema>(claim: claim, lifetime: models)
            do { try models.release(&ticket) } catch { if failure == nil { failure = error } }
            records.pointee.offeredClaims[i] = nil
        }
        if let failure { throw failure }
    }

    @inlinable
    borrowing func install(
        _ identity: Identity, ordinal: UInt16, state: inout State<Model.Schema>,
        ticket: inout ObservableModelTicket<Model.Schema>
    ) throws(ObservableStateError) {
        let key = ticket.key
        try models.validateAssociation(key)
        let slot = Int(key._giftUISlot)
        guard identities[slot] == nil else {
            throw .locationCapacityExhausted
        }
        var locations = 0
        var registrations = 0
        var associations = 0
        for index in 0 ..< 2 {
            let item = records.pointee.records[index]
            if identities[index] != nil { locations += 1 }
            if item.attachment != nil { registrations += 1 }
            if item.encountered { associations += 1 }
        }
        guard locations < Int(records.pointee.limits.maximumLocations) else {
            throw .locationCapacityExhausted
        }
        guard registrations < Int(records.pointee.limits.maximumRegistrations) else {
            throw .registrationCapacityExhausted
        }
        guard associations < Int(records.pointee.limits.maximumStagedAssociations) else {
            throw .associationStagingCapacityExhausted
        }
        let attachment = try records.pointee.reserve(UInt16(slot))
        try models.pin(key)
        records.pointee.operationActive = true
        defer { records.pointee.operationActive = false }
        identities[slot] = identity
        records.pointee.records[slot] = .init(
            ordinal: ordinal,
            keyOwner: key._giftUIOwnerInstance, keyGeneration: key._giftUIAllocationGeneration,
            attachment: attachment, encountered: true)
        reports.pointee.routes[slot] = .init()
        if let failure = reports.pointee.routes[slot].lifecycle.beginAttachment(attachment) {
            throw failure
        }
        let sink = _GiftUIObservableChangeSink(
            attachment: attachment, context: UnsafeMutableRawPointer(reports),
            reportRoute: reportObservableStateRoot)
        let returned = try models.attach(key, sink: consume sink)
        if let failure = reports.pointee.routes[slot].lifecycle.acceptAttachmentReturn(returned) {
            if returned != nil {
                do { try models.detach(key, attachment: attachment) } catch { models.requireStop() }
            }
            identities[slot] = nil
            records.pointee.records[slot] = .init()
            throw failure
        }
        try models.register(ticket)
        try bind(&state, slot: slot)
        try models.release(&ticket)
    }

    private borrowing func retire(_ slot: Int) throws(ObservableStateError) {
        guard let attachment = records.pointee.records[slot].attachment else {
            throw .invariantViolation
        }
        let former = key(at: slot)
        if let failure = reports.pointee.routes[slot].lifecycle.retire(attachment: attachment) {
            throw failure
        }
        var firstFailure: ObservableStateError?
        do { try models.detach(former, attachment: attachment) } catch { firstFailure = error }
        do { try models.unregister(former) } catch {
            if firstFailure == nil { firstFailure = error }
        }
        identities[slot] = nil
        records.pointee.records[slot] = .init()
        reports.pointee.routes[slot].dirty = false
        if let firstFailure {
            models.requireStop()
            throw firstFailure
        }
    }

    package borrowing func finishCandidate(_ disposition: ObservableStateCandidateDisposition)
        -> ObservableStateResult
    {
        do throws(ObservableStateError) {
            try checkOperation(discovery: true)
            guard models.phase == .discovering, records.pointee.candidateActive else {
                throw .invalidPhaseSafetyNotProven
            }
            records.pointee.operationActive = true
            defer {
                records.pointee.operationActive = false
                records.pointee.candidateActive = false
            }
            var changed = false
            var failure: ObservableStateError?
            for i in 0 ..< 2 where identities[i] != nil {
                let record = records.pointee.records[i]
                let remove = disposition == .discard ? !record.published : !record.encountered
                if remove {
                    do { try retire(i) } catch { if failure == nil { failure = error } }
                    changed = true
                } else {
                    if disposition == .publish {
                        changed = changed || !record.published
                        records.pointee.records[i].published = true
                        reports.pointee.routes[i].dirty = false
                    }
                    records.pointee.records[i].encountered = false
                }
            }
            do { try releaseOffers() } catch { if failure == nil { failure = error } }
            if let failure { throw failure }
            return .success(
                disposition == .discard
                    ? .candidateDiscarded : (changed ? .associationsCommitted : .unchanged))
        } catch { return .failure(error) }
    }

    package borrowing func replace(
        identity: Identity, ordinal: UInt16,
        transferring ticket: inout ObservableModelTicket<Model.Schema>, seed: Model.Schema.Seed,
        registrationAvailable: Bool = true, stagingAvailable: Bool = true
    ) -> ObservableStateResult {
        do throws(ObservableStateError) {
            try checkOperation()
            guard !records.pointee.candidateActive else { throw .reentrancyViolation }
            guard let old = matching(identity, ordinal: ordinal),
                records.pointee.records[old].published,
                key(at: old) == ticket.key
            else { throw .incompatibleAssociation }
            try models.validateTicket(ticket)
            // Target-generation exhaustion precedes retained-owner/capacity checks.
            guard records.pointee.nextGeneration != nil else {
                throw .registrationGenerationExhausted
            }
            var registrations = 0
            var associations = 0
            for index in 0 ..< 2 {
                if records.pointee.records[index].attachment != nil { registrations += 1 }
                if identities[index] != nil { associations += 1 }
            }
            let reservation = try models.stageReplacement(
                transferring: ticket, seed: seed,
                registrationAvailable: registrationAvailable
                    && registrations < Int(records.pointee.limits.maximumRegistrations),
                stagingAvailable: stagingAvailable
                    && associations < Int(records.pointee.limits.maximumStagedAssociations))
            let candidate = models.key(for: reservation.candidate)
            let new = Int(reservation.candidate.slot)
            let attachment = try records.pointee.reserve(UInt16(new))
            records.pointee.operationActive = true
            defer { records.pointee.operationActive = false }
            identities[new] = identity
            records.pointee.records[new] = .init(
                ordinal: ordinal,
                keyOwner: candidate._giftUIOwnerInstance,
                keyGeneration: candidate._giftUIAllocationGeneration,
                attachment: attachment)
            reports.pointee.routes[new] = .init()
            _ = reports.pointee.routes[new].lifecycle.beginAttachment(attachment)
            let sink = _GiftUIObservableChangeSink(
                attachment: attachment, context: UnsafeMutableRawPointer(reports),
                reportRoute: reportObservableStateRoot)
            let returned = try models.attach(candidate, sink: consume sink)
            if let failure = reports.pointee.routes[new].lifecycle.acceptAttachmentReturn(returned)
            {
                var cleanupFailed = false
                if returned != nil {
                    do { try models.detach(candidate, attachment: attachment) } catch {
                        cleanupFailed = true
                    }
                }
                identities[new] = nil
                records.pointee.records[new] = .init()
                try? models.discardReplacement(reservation)
                if cleanupFailed { models.requireStop() }
                throw failure
            }
            guard let formerAttachment = records.pointee.records[old].attachment else {
                throw .invariantViolation
            }
            // This is the commit boundary. Retire old routing before invoking its
            // detach hook; a detach invariant failure cannot roll back B.
            let retirementFailure = reports.pointee.routes[old].lifecycle.retire(
                attachment: formerAttachment)
            records.pointee.records[new].published = true
            reports.pointee.routes[new].dirty = true
            var failure = retirementFailure
            do { try models.detach(ticket.key, attachment: formerAttachment) } catch {
                if failure == nil { failure = error }
            }
            var activated = try models.commitReplacement(reservation, transferring: &ticket)
            try models.release(&activated)
            identities[old] = nil
            records.pointee.records[old] = .init()
            reports.pointee.routes[old].dirty = false
            if let failure {
                models.requireStop()
                throw failure
            }
            return .success(.replaced)
        } catch { return .failure(error) }
    }

    @inlinable
    package borrowing func mutate(_ key: _GiftUIModelKey<Model.Schema>, mutation: Model.Mutation)
        -> ObservableStateResult where Model: ObservableModelMutation
    {
        guard !records.pointee.retired else { return .failure(.invalidPhaseSafetyNotProven) }
        guard !records.pointee.operationActive else { return .failure(.reentrancyViolation) }
        reports.pointee.mutationReporting = true
        defer { reports.pointee.mutationReporting = false }
        return models.mutate(key, mutation: mutation) {
            models.report(key)
        }
    }

    @usableFromInline
    package borrowing func mutate(
        identity: Identity, ordinal: UInt16, matching generation: ObservableTargetGeneration,
        mutation: Model.Mutation
    ) -> ObservableStateResult where Model: ObservableModelMutation {
        do throws(ObservableStateError) {
            try checkOperation()
            guard let slot = matching(identity, ordinal: ordinal),
                records.pointee.records[slot].published,
                records.pointee.records[slot].attachment?.generation == generation.rawValue
            else { throw .staleModel }
            return mutate(key(at: slot), mutation: mutation)
        } catch { return .failure(error) }
    }

    @usableFromInline
    package borrowing func mutate(
        identity: Identity, ordinal: UInt16, matching generation: ObservableTargetGeneration,
        operation: () -> Model.Mutation
    ) -> ObservableStateResult where Model: ObservableModelMutation {
        do throws(ObservableStateError) {
            try checkOperation()
            guard let slot = matching(identity, ordinal: ordinal),
                records.pointee.records[slot].published,
                records.pointee.records[slot].attachment?.generation == generation.rawValue
            else { throw .staleModel }
            let key = key(at: slot)
            reports.pointee.mutationReporting = true
            defer { reports.pointee.mutationReporting = false }
            return models.mutate(key, operation: operation) { models.report(key) }
        } catch { return .failure(error) }
    }

    @usableFromInline
    package borrowing func shutdown() throws(ObservableStateError) {
        if records.pointee.retired { return }
        // A terminal activation still needs to retire all callback routes while
        // their report storage exists. Normal operations remain forbidden.
        if models.phase != .retired { try checkOperation() }
        records.pointee.operationActive = true
        defer {
            records.pointee.operationActive = false
            records.pointee.retired = true
        }
        var failure: ObservableStateError?
        for i in 0 ..< 2 where identities[i] != nil {
            do { try retire(i) } catch { if failure == nil { failure = error } }
        }
        records.pointee.candidateActive = false
        do { try releaseOffers() } catch { if failure == nil { failure = error } }
        if let failure { throw failure }
    }
}

extension ObservableStateRootScope where Model: ~Copyable {
    /// Exactly one generated lexical visit. The binder's actual root borrow
    /// cannot be copied or escaped by a public generic host witness.
    @inlinable
    @inline(__always)
    package borrowing func withBoundDeclaration<Declaration: _GiftUIObservableStateHost, Result>(
        _ declaration: borrowing Declaration, identity: Identity,
        body: (borrowing Declaration) throws(ObservableStateError) -> Result
    ) throws(ObservableStateError) -> Result {
        var transient = copy declaration
        try models.withDiscovery { () throws(ObservableStateError) in
            var visitor = ScopedObservableDeclarationVisitor(
                root: borrowedRoot(), identity: identity)
            transient._giftUIVisitObservableStateDeclarations(&visitor)
            if let failure = visitor.failure { throw failure }
        }
        return try body(transient)
    }

    /// End the body read before returning owned child declarations to traversal.
    package borrowing func evaluate<Declaration: View & _GiftUIObservableStateHost>(
        _ declaration: borrowing Declaration, identity: Identity,
        body: (borrowing Declaration, borrowing _GiftUIFrameRead<Declaration.Domain>)
            throws(_GiftUIAccessFailure) -> Declaration.Body
    ) throws(ObservableStateError) -> Declaration.Body
    where Declaration.Domain.Slot == ObservableModelReadSlot<Model> {
        try withBoundDeclaration(declaration, identity: identity) {
            (bound) throws(ObservableStateError) in
            do throws(_GiftUIAccessFailure) {
                return try models.withRead(Declaration.Domain.self) {
                    (read) throws(_GiftUIAccessFailure) in
                    try body(bound, read)
                }
            } catch { throw ObservableStateError(accessFailure: error) }
        }
    }
}

@usableFromInline
struct ScopedObservableDeclarationVisitor<
    Model: ObservableStateModel & ~Copyable, Identity: Equatable & Sendable
>: ~Copyable, ~Escapable, _GiftUIObservableStateDeclarationVisitor {
    @usableFromInline
    let root: ObservableStateRootScope<Model, Identity>
    @usableFromInline
    let identity: Identity
    @usableFromInline
    var failure: ObservableStateError?

    @_lifetime(copy root)
    @usableFromInline
    init(root: consuming ObservableStateRootScope<Model, Identity>, identity: Identity) {
        self.root = consume root
        self.identity = identity
    }

    @inlinable
    @inline(__always)
    mutating func visit<Schema: _GiftUIModelSchema>(
        _ state: inout State<Schema>, declarationOrdinal: UInt16
    ) {
        guard failure == nil else { return }
        guard Schema.self == Model.Schema.self else {
            failure = .incompatibleAssociation
            return
        }
        let result = withUnsafeMutablePointer(to: &state) { pointer in
            pointer.withMemoryRebound(to: State<Model.Schema>.self, capacity: 1) { typed in
                root.encounter(
                    identity: identity, ordinal: declarationOrdinal, state: &typed.pointee)
            }
        }
        if case .failure(let error) = result { failure = error }
    }
}

    // cannot remain live. Retire every remaining claim before destroying model
    // values, while preserving both non-reusing identity cursors.
    package mutating func finishActivation() -> UInt8 {
        var occupied: UInt8 = 0
        for index in 0 ..< Int(modelLimit) {
            if records[index].occupied { occupied |= 1 << index }
            records[index] = Record()
        }
        for index in 0 ..< Int(claimLimit) { claims[index].address = nil }
        activeRead = false
        replacement = nil
        if phase != .retired { phase = .mutating }
        return occupied
    }

    package func key<Schema>(for address: Address, as schema: Schema.Type) -> _GiftUIModelKey<
        Schema
    > {
        _GiftUIModelKey(
            ownerInstance: address.owner, allocationGeneration: address.generation,
            slot: address.slot)
    }

    @inlinable
    package func address<Schema>(for key: _GiftUIModelKey<Schema>) -> Address {
        Address(
            owner: key._giftUIOwnerInstance, generation: key._giftUIAllocationGeneration,
            slot: key._giftUISlot)
    }

    @usableFromInline
    package mutating func retireActivation() { phase = .retired }

    @usableFromInline
    package func operationAllowed(discovery: Bool = false) -> ObservableStateError? {
        if phase == .retired { return .invalidPhaseSafetyNotProven }
        if phase == .replacing || phase == .writing { return .reentrancyViolation }
        if activeRead { return .invalidPhaseSafetyNotProven }
        if phase == .mutating || (discovery && phase == .discovering) { return nil }
        return .invalidPhaseContained
    }

    private func validate(_ address: Address) throws(ObservableStateError) -> Int {
        guard address.owner == identity.rawValue, address.slot < modelLimit else {
            throw .staleModel
        }
        let index = Int(address.slot)
        guard records[index].occupied, records[index].generation == address.generation else {
            throw .staleModel
        }
        return index
    }

    private func validate(_ claim: Claim) throws(ObservableStateError) -> Int {
        let index = try validate(claim.address)
        guard claim.index < claimLimit,
            claims[Int(claim.index)].address == claim.address,
            claims[Int(claim.index)].serial == claim.serial
        else { throw .staleModel }
        return index
    }

    @usableFromInline
    package mutating func beginMutation(_ address: Address, schema: UInt16)
        throws(ObservableStateError) -> Int
    {
        if let failure = operationAllowed() { throw failure }
        let index = try validate(address)
        guard records[index].schema == schema else { throw .staleModel }
        phase = .writing
        return index
    }

    @usableFromInline
    package mutating func endMutation() {
        // Only the checked operation route can enter this phase. Keeping its
        // end non-failing avoids inventing a rollback after an admitted commit.
        if phase == .writing { phase = .mutating }
    }

    @usableFromInline
    package func validatedClaim(_ claim: Claim, schema: UInt16)
        throws(ObservableStateError) -> Int
    {
        if let failure = operationAllowed(discovery: true) { throw failure }
        let index = try validate(claim)
        guard records[index].schema == schema else { throw .staleModel }
        return index
    }

    @usableFromInline
    package func validateAssociation(_ address: Address, schema: UInt16)
        throws(ObservableStateError)
    {
        if let failure = operationAllowed(discovery: true) { throw failure }
        let index = try validate(address)
        guard records[index].schema == schema else { throw .incompatibleAssociation }
        guard !records[index].registered else { throw .duplicateOwner }
    }

    @usableFromInline
    package func callbackIndex(_ address: Address, schema: UInt16)
        throws(ObservableStateError) -> Int
    {
        let index = try validate(address)
        guard records[index].schema == schema else { throw .incompatibleAssociation }
        return index
    }

    private func freeModel() -> Int? {
        for index in 0 ..< Int(modelLimit) where !records[index].occupied { return index }
        return nil
    }

    private func freeClaim() -> Int? {
        for index in 0 ..< Int(claimLimit) where claims[index].address == nil { return index }
        return nil
    }

    private mutating func issueClaim(at index: Int, for address: Address) -> Claim {
        // Every caller reserves capacity and checks exhaustion before mutation.
        let serial = nextClaimSerial!
        let next = serial.addingReportingOverflow(1)
        nextClaimSerial = next.overflow ? nil : next.partialValue
        claims[index] = ClaimRecord(serial: serial, address: address)
        records[Int(address.slot)].claims += 1
        return Claim(address: address, index: UInt16(index), serial: serial)
    }

    @usableFromInline
    package mutating func materialize(schema: UInt16, needsCapture: Bool)
        throws(ObservableStateError) -> Claim
    {
        if let failure = operationAllowed(discovery: true) { throw failure }
        guard !generations.isExhausted else { throw .modelGenerationExhausted }
        guard let model = freeModel() else { throw .modelCapacityExhausted }
        guard let claim = freeClaim(), nextClaimSerial != nil else { throw .ownerCapacityExhausted }
        guard !needsCapture || (captureLimit == 1 && captureCount == 0) else {
            throw .modelCapacityExhausted
        }
        let generation = try generations.issue()
        let address = Address(owner: identity.rawValue, generation: generation, slot: UInt16(model))
        records[model] = Record(
            generation: generation, schema: schema, occupied: true, ownsCapture: needsCapture)
        return issueClaim(at: claim, for: address)
    }

    package mutating func retain(_ address: Address) throws(ObservableStateError) -> Claim {
        if let failure = operationAllowed() { throw failure }
        _ = try validate(address)
        guard let claim = freeClaim(), nextClaimSerial != nil else { throw .ownerCapacityExhausted }
        return issueClaim(at: claim, for: address)
    }

    // Returns the slot whose initialized model must now be destroyed by the profile.
    private mutating func reclaimIfUnowned(_ index: Int) -> UInt16? {
        guard records[index].occupied, records[index].claims == 0,
            !records[index].registered, !records[index].pinned
        else { return nil }
        records[index].occupied = false
        records[index].ownsCapture = false
        return UInt16(index)
    }

    @usableFromInline
    package mutating func release(_ claim: Claim) throws(ObservableStateError) -> UInt16? {
        if let failure = operationAllowed(discovery: true) { throw failure }
        let index = try validate(claim)
        claims[Int(claim.index)].address = nil
        records[index].claims -= 1
        return reclaimIfUnowned(index)
    }

    @usableFromInline
    package mutating func register(_ address: Address, schema: UInt16) throws(ObservableStateError)
    {
        if let failure = operationAllowed(discovery: true) { throw failure }
        let index = try validate(address)
        guard records[index].schema == schema else { throw .incompatibleAssociation }
        guard !records[index].registered else { throw .duplicateOwner }
        records[index].registered = true
    }

    package mutating func unregister(_ address: Address) throws(ObservableStateError) -> UInt16? {
        if let failure = operationAllowed(discovery: true) { throw failure }
        let index = try validate(address)
        guard records[index].registered else { throw .staleAttachment }
        records[index].registered = false
        return reclaimIfUnowned(index)
    }

    package mutating func beginFrame() throws(ObservableStateError) {
        if let failure = operationAllowed() { throw failure }
        phase = .reading
    }

    @usableFromInline
    package mutating func beginDiscovery() throws(ObservableStateError) {
        guard phase != .replacing else { throw .reentrancyViolation }
        guard phase == .reading, !activeRead else { throw .invalidPhaseSafetyNotProven }
        phase = .discovering
    }

    @usableFromInline
    package mutating func endDiscovery() throws(ObservableStateError) {
        guard phase == .discovering else { throw .invalidPhaseSafetyNotProven }
        phase = .reading
    }

    @usableFromInline
    package mutating func pin(_ address: Address) throws(ObservableStateError) {
        guard phase != .replacing else { throw .reentrancyViolation }
        guard phase == .reading || phase == .discovering, !activeRead else {
            throw .invalidPhaseSafetyNotProven
        }
        let index = try validate(address)
        if records[index].pinned { return }
        guard pinCount < Int(pinLimit) else { throw .ownerCapacityExhausted }
        records[index].pinned = true
    }

    @usableFromInline
    package mutating func beginRead() throws(ObservableStateError) {
        guard phase == .reading, !activeRead else { throw .invalidPhaseSafetyNotProven }
        activeRead = true
    }

    package func bind<Schema>(_ key: _GiftUIModelKey<Schema>, schema: UInt16)
        throws(_GiftUIAccessFailure) -> Int
    {
        guard phase == .reading, activeRead else { throw .invalidPhase }
        guard key._giftUIOwnerInstance == identity.rawValue else { throw .wrongOwner }
        guard key._giftUISlot < modelLimit else { throw .staleModel }
        let index = Int(key._giftUISlot)
        guard records[index].schema == schema else { throw .wrongSchema }
        guard records[index].occupied, records[index].generation == key._giftUIAllocationGeneration
        else { throw .staleModel }
        guard records[index].pinned else { throw .invalidPhase }
        return index
    }

    @usableFromInline
    package mutating func endRead() throws(ObservableStateError) {
        guard phase == .reading, activeRead else { throw .invalidPhaseSafetyNotProven }
        activeRead = false
    }

    @usableFromInline
    package mutating func endFrame(reclaim: (UInt16) -> Void) throws(ObservableStateError) {
        guard phase == .reading, !activeRead else { throw .invalidPhaseSafetyNotProven }
        for index in 0 ..< Int(modelLimit) {
            records[index].pinned = false
            if let slot = reclaimIfUnowned(index) { reclaim(slot) }
        }
        phase = .mutating
    }

    package mutating func reserveReplacement(
        transferring finalClaim: Claim, schema: UInt16, candidateIsEmpty: Bool,
        registrationAvailable: Bool, stagingAvailable: Bool
    ) throws(ObservableStateError) -> Replacement {
        if let failure = operationAllowed() { throw failure }
        let old = try validate(finalClaim)
        guard records[old].schema == schema, records[old].registered else {
            throw .incompatibleAssociation
        }
        guard !generations.isExhausted else { throw .modelGenerationExhausted }
        guard records[old].claims == 1, !records[old].pinned else { throw .retainedModel }
        guard candidateIsEmpty else { throw .modelCapacityExhausted }
        guard let model = freeModel() else { throw .modelCapacityExhausted }
        guard let claim = freeClaim(), nextClaimSerial != nil else { throw .ownerCapacityExhausted }
        guard registrationAvailable else { throw .registrationCapacityExhausted }
        guard stagingAvailable else { throw .replacementStagingCapacityExhausted }
        let generation = try generations.issue()
        let address = Address(owner: identity.rawValue, generation: generation, slot: UInt16(model))
        records[model] = Record(generation: generation, schema: schema, occupied: true)
        let candidate = issueClaim(at: claim, for: address)
        let reservation = Replacement(
            former: finalClaim.address, candidate: address, formerClaim: finalClaim,
            candidateClaim: candidate)
        replacement = reservation
        phase = .replacing
        return reservation
    }

    package mutating func finishReplacement(_ reservation: Replacement, commit: Bool)
        throws(ObservableStateError) -> Claim?
    {
        guard phase == .replacing, let current = replacement,
            current.formerClaim == reservation.formerClaim,
            current.candidateClaim == reservation.candidateClaim
        else { throw .invariantViolation }
        let old = try validate(reservation.formerClaim)
        let new = try validate(reservation.candidateClaim)
        if commit {
            records[new].registered = true
            records[new].ownsCapture = records[old].ownsCapture
            records[old].registered = false
            records[old].ownsCapture = false
            claims[Int(reservation.formerClaim.index)].address = nil
            records[old].claims = 0
            records[old].occupied = false
        } else {
            claims[Int(reservation.candidateClaim.index)].address = nil
            records[new].claims = 0
            records[new].occupied = false
        }
        replacement = nil
        phase = .mutating
        return commit ? reservation.candidateClaim : nil
    }
}
